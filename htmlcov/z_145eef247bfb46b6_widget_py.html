10def mask_account_card(info: str) -> str: 

11 """Маскирует номер карты или счета в строке вида "Тип Номер". 

12 

13 Строка приходит одним аргументом, например: 

14 "Visa Platinum 7000792289606361" 

15 "Maestro 7000792289606361" 

16 "Счет 73654108430135874305" 

17 

18 Внутри строка разбирается на название и номер (это не два отдельных 

19 аргумента функции, а внутренняя логика), после чего номер маскируется 

20 подходящей функцией — get_mask_card_number для карт или 

21 get_mask_account для счетов. 

22 

23 Пример: 

24 >>> mask_account_card("Visa Platinum 7000792289606361") 

25 'Visa Platinum 7000 79** **** 6361' 

26 >>> mask_account_card("Счет 73654108430135874305") 

27 'Счет **4305' 

28 

29 Args: 

30 info: строка вида "<тип> <номер>". 

31 

32 Returns: 

33 Строка с тем же типом и замаскированным номером. 

34 """ 

35 name, number = info.rsplit(maxsplit=1) 

36 if "счет" in name.lower(): 

37 masked_number = get_mask_account(number) 

38 else: 

39 masked_number = get_mask_card_number(number) 

40 return f"{name} {masked_number}"

43def get_date(date_string: str) -> str: 

44 """Преобразует дату из ISO-формата в формат "ДД.ММ.ГГГГ". 

45 

46 Пример: 

47 >>> get_date("2024-03-11T02:26:18.671407") 

48 '11.03.2024' 

49 

50 Args: 

51 date_string: дата в формате ISO 8601, 

52 например "2024-03-11T02:26:18.671407". 

53 

54 Returns: 

55 Дата в формате "ДД.ММ.ГГГГ". 

56 """ 

57 if not isinstance(date_string, str): 

58 raise TypeError("Дата должна быть строкой") 

59 parsed_date = datetime.fromisoformat(date_string.replace("Z", "+00:00")) 

60 return parsed_date.strftime("%d.%m.%Y")

63def format_transaction(transaction: Dict[str, Any]) -> str: 

64 """Форматирует дату, описание, маскированные реквизиты и сумму операции. 

65 

66 Для открытия вклада без отправителя выводится только получатель. 

67 RUB обозначается как «руб.», остальные валюты — своим кодом. 

68 Сумма выводится без незначащих нулей, без конвертации валют. 

69 """ 

70 heading = f"{get_date(transaction['date'])} {transaction['description']}" 

71 destination = mask_account_card(transaction["to"]) 

72 source = transaction.get("from") 

73 route = f"{mask_account_card(source)} -> {destination}" if source else destination 

74 operation_amount = transaction["operationAmount"] 

75 amount = format(Decimal(str(operation_amount["amount"])), "f") 

76 if "." in amount: 

77 amount = amount.rstrip("0").rstrip(".") 

78 code = operation_amount["currency"]["code"] 

79 currency = "руб." if code == "RUB" else code 

80 return f"{heading}\n{route}\nСумма: {amount} {currency}"
