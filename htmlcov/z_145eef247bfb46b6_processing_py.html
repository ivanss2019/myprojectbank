7def filter_by_state( 

8 data: List[Dict[str, Any]], state: str = "EXECUTED" 

9) -> List[Dict[str, Any]]: 

10 """Отбирает операции с заданным статусом. 

11 

12 Порядок элементов, соответствующих условию, сохраняется таким же, 

13 как в исходном списке. 

14 

15 Пример: 

16 >>> operations = [ 

17 ... {"id": 1, "state": "EXECUTED", "date": "2019-07-03T18:35:29"}, 

18 ... {"id": 2, "state": "CANCELED", "date": "2018-09-12T21:27:25"}, 

19 ... ] 

20 >>> filter_by_state(operations, "CANCELED") 

21 [{'id': 2, 'state': 'CANCELED', 'date': '2018-09-12T21:27:25'}] 

22 

23 Args: 

24 data: список словарей с данными операций. 

25 state: значение ключа "state", по которому фильтруются операции. 

26 По умолчанию "EXECUTED". 

27 

28 Returns: 

29 Новый список словарей, содержащий только операции с указанным 

30 статусом. 

31 """ 

32 return [item for item in data if item.get("state") == state]

35def sort_by_date( 

36 data: List[Dict[str, Any]], descending: bool = True 

37) -> List[Dict[str, Any]]: 

38 """Сортирует операции по дате. 

39 

40 Пример: 

41 >>> operations = [ 

42 ... {"id": 1, "date": "2018-06-30T02:08:58"}, 

43 ... {"id": 2, "date": "2019-07-03T18:35:29"}, 

44 ... ] 

45 >>> sort_by_date(operations) 

46 [{'id': 2, 'date': '2019-07-03T18:35:29'}, 

47 {'id': 1, 'date': '2018-06-30T02:08:58'}] 

48 

49 Args: 

50 data: список словарей с данными операций, каждый из которых 

51 содержит ключ "date" с датой в формате ISO 8601. 

52 descending: порядок сортировки. True (по умолчанию) — от самой 

53 новой операции к самой старой, False — наоборот. 

54 

55 Returns: 

56 Новый список словарей, отсортированный по дате. 

57 """ 

58 return sorted( 

59 data, 

60 key=lambda item: datetime.fromisoformat(item["date"]), 

61 reverse=descending, 

62 )
